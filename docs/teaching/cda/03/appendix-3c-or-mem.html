library(haven)           # import Stata data and labels
library(dplyr)           # data manipulation
library(marginaleffects) # predictions and marginal effects
library(questionr)       # odds ratios

# Health and Retirement Study 2006
hrs <- read_dta("https://tdmize.github.io/data/data/hrs06.dta")

# Treat binary/nominal variables as factors using their value labels
hrs <- hrs |>
  mutate(black = as_factor(black), woman = as_factor(woman), ed4cat = as_factor(ed4cat))

diabmod <- glm(diabetes ~ age + black + bmi + woman + ed4cat,
  family = binomial(link = "logit"), data = hrs)
odds.ratio(diabmod, level = .95, digits = 4)
